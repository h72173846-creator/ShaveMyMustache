========================================
كيف تعمل APK و AAB للعبة
احلق شنبي لو كسبت! 🥸
========================================

المتطلبات:
1. Godot 4.2 أو أحدث (يفضل 4.3)
2. Android Studio (أو على الأقل Android SDK + Command-line tools)
3. JDK 17

خطوات بناء الـ APK (للاختبار):

1. افتح المشروع في Godot
2. روح على قائمة:
   Project → Export...
3. اضغط "Add..." واختار Android
4. في خانة "Debug Keystore":
   - لو مش عندك، اضغط "Create" وخلق واحد جديد
5. حط Package Name مثلاً:
   com.yourname.shavemustache
6. اضغط "Export Project"
7. اختار مكان الحفظ واكتب اسم الملف (مثل ShaveMustache.apk)
8. استنى لما يخلص

خطوات الـ AAB (للرفع على Google Play):

1. نفس الخطوات فوق
2. بدل ما تختار Export Project، اختار "Export AAB"
3. الملف الناتج يكون .aab
4. ارفعه على Google Play Console

ملاحظات مهمة جداً:

- قبل النشر النهائي لازم تعمل Keystore خاص بالـ Release وتحفظه في مكان آمن
- لازم تعمل أيقونة أصلية (1024x1024) وتضعها في المشروع
- لازم تضيف صور شاشة (Screenshots) للمتجر
- سياسة خصوصية: لو اللعبة مش بتجمع بيانات، تقدر تكتب سياسة بسيطة

حجم اللعبة الحالي صغير جداً (ممتاز للأجهزة الضعيفة).

بعد ما تبني الـ APK:
- انقله على الموبايل وثبته (فعل "مصادر غير معروفة" لو لزم)
- جرب كل المراحل وتأكد إن الأزرار شغالة كويس

لو واجهت مشكلة في التصدير، ابعتلي رسالة الخطأ بالظبط.

# احلق شنبي لو كسبت! 🥸

لعبة منصات 2D عربية أصلية مبنية بمحرك Godot 4، وأولوية التطوير لنسخة Android.

## محتويات المشروع الحالية
- 9 ملفات مراحل (Level01 إلى Level09).
- حركة اللاعب والقفز مع دعم لوحة المفاتيح وأزرار اللمس.
- شاشة رئيسية واختيار المراحل وحفظ التقدم.
- فخاخ: منصات مختفية، باب وهمي، سقف ساقط، أسهم مضللة، وجاذبية معكوسة.
- إعدادات صوت رئيسية تحفظ اختيار اللاعب.

## التشغيل
1. افتح المشروع باستخدام Godot 4.x.
2. اختر Import وحدد مجلد `ShaveMyMustache`.
3. شغّل المشروع بزر F6 أو F5.

## التحكم
- الكمبيوتر: A/D أو الأسهم للحركة، وSpace أو W للقفز.
- الموبايل: أزرار يسار ويمين وقفز على الشاشة.

## ملاحظات الاختبار
تم إصلاح مراجع العقد المفقودة، ومنطقة الجاذبية، ومناطق اللمس، وعدد خانات اختيار المراحل.
لم يتم تنفيذ تشغيل فعلي للمشروع داخل Godot في بيئة إعداد هذه النسخة؛ لذلك يلزم فتحه في Godot واختباره قبل اعتبار APK/AAB جاهزًا للنشر.

## النشر
إعداد Android Export في Godot يتطلب قوالب التصدير وAndroid SDK/JDK، ثم إنشاء APK للاختبار أو AAB للنشر على Google Play بعد استيفاء متطلبات المتجر.


## متجر السكنات والكوينز - إصدار أولي
- يحصل اللاعب على 20 كوين عند إنهاء المرحلة.
- يحصل على 5 كوين إضافية إذا أنهى المرحلة دون موت.
- الرصيد والسكنات المملوكة والسكن المجهز تُحفظ محليًا.
- يوجد متجر داخل اللعبة لشراء وتجهيز السكنات.
- السكنات الحالية شخصيات أصلية مؤقتة: قط أزرق، متحوّل أخضر، نينجا برتقالي، نينجا أزرق.
- يتغير لون الشخصية داخل اللعب حسب السكن المجهز.

### حقوق الشخصيات المعروفة
جامبول وبن تن وناروتو وبوروتو شخصيات محمية بحقوق ملكية فكرية. لا تستخدم أسماءها أو تصميماتها الأصلية أو صورها في نسخة منشورة على Google Play إلا بعد الحصول على الترخيص اللازم. السكنات الأصلية المؤقتة هنا بدائل مستقلة وليست تلك الشخصيات.

extends Control

@onready var play_button: Button = $PlayButton
@onready var levels_button: Button = $LevelsButton
@onready var settings_button: Button = $SettingsButton
@onready var deaths_label: Label = $DeathsLabel

var game_manager: GameManager
var settings_popup: PopupPanel
var volume_slider: HSlider
var coins_label: Label

func _ready() -> void:
    game_manager = get_node_or_null("/root/GameManager") as GameManager
    play_button.pressed.connect(_on_play_pressed)
    levels_button.pressed.connect(_on_levels_pressed)
    settings_button.pressed.connect(_on_settings_pressed)
    _build_settings_popup()
    _build_shop_entry()
    _update_deaths()
    if game_manager:
        game_manager.coins_changed.connect(_update_coins)
    _update_coins()

func _update_deaths() -> void:
    if game_manager and deaths_label:
        deaths_label.text = "عدد مرات الموت: " + str(game_manager.total_deaths)

func _on_play_pressed() -> void:
    if game_manager:
        game_manager.start_level(game_manager.current_level)
    else:
        get_tree().change_scene_to_file("res://levels/Level01.tscn")

func _on_levels_pressed() -> void:
    get_tree().change_scene_to_file("res://scenes/LevelSelect.tscn")

func _on_settings_pressed() -> void:
    if settings_popup:
        settings_popup.popup_centered(Vector2i(420, 220))

func _build_settings_popup() -> void:
    settings_popup = PopupPanel.new()
    settings_popup.name = "SettingsPopup"
    add_child(settings_popup)
    var panel := VBoxContainer.new()
    panel.custom_minimum_size = Vector2(360, 150)
    panel.add_theme_constant_override("separation", 14)
    settings_popup.add_child(panel)

var facing_right: bool = true

# Mobile input
var mobile_left: bool = false
var mobile_right: bool = false
var mobile_jump: bool = false

@onready var visual: Node2D = $Visual

	add_to_group("player")

func _physics_process(delta: float) -> void:
	if is_dead or not can_move:
		return

	# Gravity supports normal and inverted gravity zones.
	var gravity_value := gravity * gravity_multiplier
	up_direction = Vector2.UP if gravity_multiplier >= 0.0 else Vector2.DOWN
	if not is_on_floor():
		velocity.y += gravity_value * delta
		velocity.y = clampf(velocity.y, -max_fall_speed, max_fall_speed)
		coyote_timer = maxf(0.0, coyote_timer - delta)
	else:
		coyote_timer = coyote_time

	# Jump buffer
	if jump_buffer_timer > 0:
		jump_buffer_timer -= delta

	# Horizontal movement
	var direction := 0.0
	if mobile_left or Input.is_action_pressed("move_left"):
		direction -= 1.0
	if mobile_right or Input.is_action_pressed("move_right"):
		direction += 1.0

	var is_running = false
	if direction != 0:
		velocity.x = direction * speed
		facing_right = direction > 0
		is_running = true
		if visual and visual.has_method("set_facing"):
			visual.set_facing(facing_right)
	else:
		velocity.x = move_toward(velocity.x, 0, speed * 9.0 * delta)

	# Jump
	var mobile_jump_just_pressed := mobile_jump and not mobile_jump_was_pressed
	mobile_jump_was_pressed = mobile_jump
	var wants_jump := mobile_jump_just_pressed or Input.is_action_just_pressed("jump")
	if wants_jump:
		jump_buffer_timer = jump_buffer_time

	if jump_buffer_timer > 0 and coyote_timer > 0:
		velocity.y = -jump_velocity if gravity_multiplier < 0.0 else jump_velocity
		coyote_timer = 0.0
		jump_buffer_timer = 0.0

	# Update visual state
	if visual and visual.has_method("set_state"):
		visual.set_state(is_running and is_on_floor(), not is_on_floor() and velocity.y < 0, not is_on_floor() and velocity.y > 50)

	move_and_slide()

	# Check hazards
	for i in get_slide_collision_count():
		var col = get_slide_collision(i)
		var collider = col.get_collider()
		if collider and collider.is_in_group("hazard"):
			die()

func die() -> void:
	if is_dead:
		return
	is_dead = true
	can_move = false
	velocity = Vector2.ZERO
	if death_particles:
		death_particles.emitting = true
	if visual:
		visual.modulate = Color(1, 0.3, 0.3)
	died.emit()

func reset_state() -> void:
	is_dead = false
	can_move = true
	velocity = Vector2.ZERO
	coyote_timer = 0.0
	jump_buffer_timer = 0.0
	mobile_left = false
	mobile_right = false
	mobile_jump = false
	mobile_jump_was_pressed = false
	set_gravity_multiplier(1.0)
	if visual:
		visual.modulate = Color.WHITE

func set_mobile_input(left: bool, right: bool, jump: bool) -> void:
	mobile_left = left
	mobile_right = right
	mobile_jump = jump

func set_gravity_multiplier(value: float) -> void:
	gravity_multiplier = value if not is_zero_approx(value) else 1.0
	if visual:
		visual.scale.y = -1.0 if gravity_multiplier < 0.0 else 1.0

        return
    var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
    if file == null:
        return
    var data = file.get_var()
    file.close()
    if not data is Dictionary:
        return

    total_deaths = int(data.get("total_deaths", 0))
    current_level = int(data.get("current_level", 1))

    if not owns_skin(equipped_skin):
        equipped_skin = "default"

    status_label.text = "السكن الحالي: " + str(manager.skins[manager.equipped_skin].name)
    for child in list.get_children():
        child.queue_free()
    for skin_id in manager.skins.keys():
        var data: Dictionary = manager.skins[skin_id]
        var row := HBoxContainer.new()
        row.custom_minimum_size.y = 72
        list.add_child(row)
        var swatch := ColorRect.new()
        swatch.custom_minimum_size = Vector2(48, 48)
        swatch.color = data.get("body", Color.WHITE)
        row.add_child(swatch)

        details.size_flags_horizontal = Control.SIZE_EXPAND_FILL
        row.add_child(details)
        var name_label := Label.new()
        name_label.text = str(data.name)
        name_label.add_theme_font_size_override("font_size", 18)
        details.add_child(name_label)
        var description := Label.new()
        description.text = str(data.description)
        description.add_theme_font_size_override("font_size", 12)
        details.add_child(description)
        var button := Button.new()
        button.custom_minimum_size = Vector2(118, 52)
        if manager.equipped_skin == skin_id:
            button.text = "مجهّز"
            button.disabled = true
        elif manager.owns_skin(skin_id):
            button.text = "تجهيز"
            button.pressed.connect(_on_skin_action.bind(skin_id))
        else:
            button.text = "شراء %d" % int(data.price)
            button.disabled = manager.coins < int(data.price)
            button.pressed.connect(_on_skin_action.bind(skin_id))
        row.add_child(button)

func _on_skin_action(skin_id: String) -> void:
    if manager.buy_or_equip_skin(skin_id):
        status_label.text = "تم تجهيز السكن!"
    else:
        status_label.text = "كوينزك مش كفاية! العب واجمع كوينز."

extends AnimatableBody2D
class_name FallingCeiling

## سقف شيطاني - بيسقط بسرعة مرعبة

@export var fall_delay: float = 0.12
@export var fall_speed: float = 750.0
@export var trigger_height: float = 100.0
@export var reset_time: float = 2.2
@export var one_shot: bool = false
@export var crush_damage: bool = true

var original_pos: Vector2
var is_falling: bool = false
var has_fallen: bool = false
var player: Player = null

@onready var collision: CollisionShape2D = $CollisionShape2D
@onready var timer: Timer = $Timer
@onready var color_rect: ColorRect = $ColorRect

func _ready() -> void:
	original_pos = global_position
	timer.one_shot = true
	add_to_group("hazard")

func _physics_process(delta: float) -> void:
	if has_fallen and one_shot:
		return
	
	player = get_tree().get_first_node_in_group("player") as Player
	if player == null:
		return
	
	var horizontal_close = abs(player.global_position.x - global_position.x) < 100
	var under = player.global_position.y > global_position.y and player.global_position.y < global_position.y + trigger_height + 70
	
	if horizontal_close and under and not is_falling and not has_fallen:
		_start_fall()
	
	if is_falling:
		global_position.y += fall_speed * delta
		# يقتل لو لمس اللاعب وهو نازل
		if crush_damage and player and global_position.distance_to(player.global_position) < 55:
			player.die()
		if global_position.y > original_pos.y + 450:
			_reset()

func _start_fall() -> void:
	is_falling = true
	# تأخير بسيط جدًا عشان اللاعب يحس إنه كان ممكن ينجو
	await get_tree().create_timer(fall_delay).timeout

func _reset() -> void:
	is_falling = false
	has_fallen = true
	global_position = original_pos
	if not one_shot:
		await get_tree().create_timer(reset_time).timeout
		has_fallen = false

## منصة خائنة - تختفي أسرع وأقسى

enum TriggerMode { ON_PLAYER_NEAR, ON_PLAYER_STAND, TIMED }

@export var trigger_mode: TriggerMode = TriggerMode.ON_PLAYER_NEAR
@export var disappear_delay: float = 0.18   # أسرع بكتير
@export var reappear_time: float = 2.8
@export var detection_distance: float = 95.0
@export var one_shot: bool = false
@export var fake_safe_time: float = 0.0     # تظهر آمنة شوية بعدين تختفي فجأة

var is_disappeared: bool = false
var player: Player = null
var has_triggered: bool = false
var stood_time: float = 0.0

@onready var collision: CollisionShape2D = $CollisionShape2D
@onready var sprite: Sprite2D = $Sprite2D
@onready var timer: Timer = $Timer
@onready var color_rect: ColorRect = $ColorRect

func _ready() -> void:
	timer.one_shot = true
	timer.timeout.connect(_on_timer_timeout)
	add_to_group("platform")

func _physics_process(delta: float) -> void:
	if is_disappeared or (has_triggered and one_shot):
		return
	
	player = get_tree().get_first_node_in_group("player") as Player
	if player == null:
		return
	
	match trigger_mode:
		TriggerMode.ON_PLAYER_NEAR:
			var dist = global_position.distance_to(player.global_position)
			if dist < detection_distance and player.global_position.y < global_position.y + 50:
				_start_disappear()
		TriggerMode.ON_PLAYER_STAND:
			if player.is_on_floor():
				for i in range(player.get_slide_collision_count()):
					if player.get_slide_collision(i).get_collider() == self:
						_start_disappear()

func _start_disappear() -> void:
	if has_triggered:
		return
	has_triggered = true
	timer.start(disappear_delay)

func _on_timer_timeout() -> void:
	if not is_disappeared:
		is_disappeared = true
		collision.set_deferred("disabled", true)
		var tween = create_tween()
		if color_rect:
			tween.tween_property(color_rect, "modulate:a", 0.0, 0.08)
		elif sprite:
			tween.tween_property(sprite, "modulate:a", 0.0, 0.08)
		
		if reappear_time > 0:
			await get_tree().create_timer(reappear_time).timeout
			_reappear()

func _reappear() -> void:
	is_disappeared = false
	collision.set_deferred("disabled", false)
	var tween = create_tween()
	if color_rect:
		tween.tween_property(color_rect, "modulate:a", 1.0, 0.15)
	elif sprite:
		tween.tween_property(sprite, "modulate:a", 1.0, 0.15)
	if not one_shot:
		has_triggered = false

func player_stood_on() -> void:
	if trigger_mode == TriggerMode.ON_PLAYER_STAND:
		_start_disappear()

@export var run_speed: float = 420.0

@export var fake_message: String = "مبروك!!! 🎉 ... لأ 😈"

var original_pos: Vector2
var player: Player = null

@onready var sprite: Sprite2D = $Sprite2D
@onready var label: Label = $Label
@onready var color_rect: ColorRect = $ColorRect

func _ready() -> void:
	original_pos = global_position
	body_entered.connect(_on_body_entered)
	if label:
		label.visible = false
		label.text = fake_message

func _on_body_entered(body: Node2D) -> void:
	if has_triggered:
		return
	if body is Player or body.is_in_group("player"):
		has_triggered = true
		player = body as Player
		_activate_trap()

func _activate_trap() -> void:
	match behavior:
		FakeBehavior.RUN_AWAY:
			_run_away()
		FakeBehavior.DISAPPEAR:
			_disappear()
		FakeBehavior.TURN_INTO_HAZARD:
			_turn_hazard()
		FakeBehavior.FAKE_WIN_MESSAGE:
			_show_fake_message()
		FakeBehavior.TELEPORT:
			_teleport()

func _run_away() -> void:
	var dir = 1.0 if player and player.global_position.x < global_position.x else -1.0
	# أحيانًا يطير لفوق كمان
	var up = -60.0 if randf() > 0.4 else 0.0
	var target = global_position + Vector2(dir * run_distance, up)
	var tween = create_tween()
	tween.set_ease(Tween.EASE_OUT)
	tween.set_trans(Tween.TRANS_BACK)
	tween.tween_property(self, "global_position", target, 0.55)

func _disappear() -> void:
	var tween = create_tween()
	if color_rect:
		tween.tween_property(color_rect, "modulate:a", 0.0, 0.15)
	tween.tween_callback(queue_free)

func _turn_hazard() -> void:
	if color_rect:
		color_rect.color = Color(1.0, 0.15, 0.15)
	add_to_group("hazard")
	# يقتل فورًا
	if player:
		player.die()

func _show_fake_message() -> void:
	if label:
		label.visible = true
		label.text = fake_message
	await get_tree().create_timer(0.9).timeout
	if label:
		label.visible = false
	_run_away()

func _teleport() -> void:
	var tween = create_tween()
	if color_rect:
		tween.tween_property(color_rect, "modulate:a", 0.0, 0.1)
	await tween.finished
	global_position = original_pos + teleport_offset
	if color_rect:
		var t2 = create_tween()
		t2.tween_property(color_rect, "modulate:a", 1.0, 0.15)
	has_triggered = false  # يقدر يخدع تاني
